-- 1. Create formulations table if it does not exist
CREATE TABLE IF NOT EXISTS formulations (
    id SERIAL PRIMARY KEY,
    product_code TEXT UNIQUE,
    scent_name TEXT,
    size_value TEXT,
    oil_id TEXT,
    oil_percentage NUMERIC(5, 2) DEFAULT 20,
    solvent_id TEXT,
    solvent_percentage NUMERIC(5, 2) DEFAULT 80,
    bottle_id TEXT,
    bottle_sticker_id TEXT,
    box_id TEXT,
    box_sticker_id TEXT,
    wrapping_id TEXT,
    price NUMERIC(12, 2),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);
