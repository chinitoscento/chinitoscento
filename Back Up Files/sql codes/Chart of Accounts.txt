-- 1. Chart of Accounts Table
CREATE TABLE IF NOT EXISTS chart_of_accounts (
    account_code VARCHAR(50) PRIMARY KEY,
    account_name VARCHAR(255) NOT NULL,
    account_type VARCHAR(50) NOT NULL CHECK (account_type IN ('Asset', 'Liability', 'Equity', 'Revenue', 'Expense')),
    sub_category VARCHAR(100),
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 2. Journal Entries Header Table
CREATE TABLE IF NOT EXISTS journal_entries (
    id SERIAL PRIMARY KEY,
    je_number VARCHAR(100) UNIQUE NOT NULL,
    date DATE NOT NULL DEFAULT CURRENT_DATE,
    reference_no VARCHAR(100),
    description TEXT,
    status VARCHAR(50) DEFAULT 'Posted',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 3. Journal Entry Line Items Table
CREATE TABLE IF NOT EXISTS journal_entry_lines (
    id SERIAL PRIMARY KEY,
    je_number VARCHAR(100) REFERENCES journal_entries(je_number) ON DELETE CASCADE,
    account_code VARCHAR(50) REFERENCES chart_of_accounts(account_code),
    description TEXT,
    debit NUMERIC(12, 2) DEFAULT 0.00,
    credit NUMERIC(12, 2) DEFAULT 0.00,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- Enable RLS
ALTER TABLE chart_of_accounts ENABLE ROW LEVEL SECURITY;
ALTER TABLE journal_entries ENABLE ROW LEVEL SECURITY;
ALTER TABLE journal_entry_lines ENABLE ROW LEVEL SECURITY;

-- Drop existing policies first to prevent re-run errors
DROP POLICY IF EXISTS "Enable all access for chart_of_accounts" ON chart_of_accounts;
DROP POLICY IF EXISTS "Enable all access for journal_entries" ON journal_entries;
DROP POLICY IF EXISTS "Enable all access for journal_entry_lines" ON journal_entry_lines;

-- Recreate policies safely
CREATE POLICY "Enable all access for chart_of_accounts" ON chart_of_accounts FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Enable all access for journal_entries" ON journal_entries FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Enable all access for journal_entry_lines" ON journal_entry_lines FOR ALL USING (true) WITH CHECK (true);
